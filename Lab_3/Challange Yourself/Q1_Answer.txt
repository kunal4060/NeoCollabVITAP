import java.util.Scanner;
public class Main{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
if(sc.hasNextInt()){
int r=sc.nextInt();
int c=sc.nextInt();
int[][] m=new int[r][c];
for(int i=0;i<r;i++)
for(int j=0;j<c;j++)
m[i][j]=sc.nextInt();
for(int i=0;i<r;i++)
for(int j=1;j<c;j+=2)
m[i][j]=m[i][j]*m[i][j];
for(int i=0;i<r;i++){
int last=m[i][c-1];
for(int j=c-1;j>0;j--)
m[i][j]=m[i][j-1];
m[i][0]=last;
}
for(int i=0;i<r;i++){
for(int j=0;j<c;j++){
System.out.print(m[i][j]);
if(j<c-1) System.out.print(" ");
}
System.out.println();
}
}
}
}
