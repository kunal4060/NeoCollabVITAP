import java.util.Scanner;
public class Main{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
if(!sc.hasNextInt())
return;
int n=sc.nextInt();
int[] arr=new int[n];
for(int i=0;i<n;i++)
arr[i]=sc.nextInt();
for(int i=0;i<n-1;i+=2){
int temp=arr[i];
arr[i]=arr[i+1];
arr[i+1]=temp;
}
int start=0;
int end=n-1;
while(start<end){
int temp=arr[start];
arr[start]=arr[end];
arr[end]=temp;
start++;
end--;
}
for(int i=0;i<n;i++)
System.out.print(arr[i]+(i==n-1?"":" "));
}
}
