import java.util.Scanner;
public class Main{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
if(!sc.hasNextInt())
return;
int n=sc.nextInt();
int[][] matrix=new int[n][n];
for(int i=0;i<n;i++)
for(int j=0;j<n;j++)
matrix[i][j]=sc.nextInt();
long mainDiagonalSumOfSquares=0;
long antiDiagonalSumOfSquares=0;
for(int i=0;i<n;i++){
long mainVal=matrix[i][i];
mainDiagonalSumOfSquares+=mainVal*mainVal;
long antiVal=matrix[i][n-1-i];
antiDiagonalSumOfSquares+=antiVal*antiVal;
}
System.out.println("Square of the first diagonal: "+mainDiagonalSumOfSquares);
System.out.println("Square of the second diagonal: "+antiDiagonalSumOfSquares);
}
}
