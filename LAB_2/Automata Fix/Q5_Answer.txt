import java.util.Scanner;
class Main {
static int[] gradeStack = new int[50];
static int gradeTop = -1;
static void pushGrade(int grade) {
gradeStack[++gradeTop] = grade;
}
static void peekGrade() {
if (gradeTop == -1) {
return;
}
System.out.println(gradeStack[gradeTop]);
}
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
if(sc.hasNextInt()) {
int n = sc.nextInt();
for (int i = 0; i < n; i++) {
int grade = sc.nextInt();
pushGrade(grade);
}
peekGrade();
}
sc.close();
}
}
