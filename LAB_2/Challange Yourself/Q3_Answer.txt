import java.util.Scanner;
public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
if (!sc.hasNextInt()) return;
int n1 = sc.nextInt();
int[] arr = new int[100];
int top1 = -1;
int top2 = 100;
for (int i = 0; i < n1; i++) {
arr[++top1] = sc.nextInt();
}
int n2 = sc.nextInt();
for (int i = 0; i < n2; i++) {
arr[--top2] = sc.nextInt();
}
System.out.print("Stack1: ");
for (int i = top1; i >= 0; i--) {
System.out.print(arr[i] + (i == 0 ? "" : " "));
}
System.out.println();
System.out.print("Stack2: ");
for (int i = top2; i < 100; i++) {
System.out.print(arr[i] + (i == 99 ? "" : " "));
}
System.out.println();
sc.close();
}
}
