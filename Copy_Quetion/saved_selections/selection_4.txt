// You are using Java
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int pos = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 0) {
                arr[pos] = arr[i];
                pos++;
            }
        }
        while (pos > n) {
            arr[pos] = 1;
            pos--;
        }

        for (int i = 0; i < n; i++) {
            if (i > 0) System.out.print(" ");
            System.out.print(arr[i]);
        }
        System.out.println();
    }
}
