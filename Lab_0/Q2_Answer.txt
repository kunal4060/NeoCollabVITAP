import java.util.*;

public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

int n = sc.nextInt();
int[] arr = new int[n];
for (int i = 0; i < n; i++)
arr[i] = sc.nextInt();

int lo = 0, hi = n - 1;
while (lo < hi) {
int mid = (lo + hi) / 2;
if (arr[mid] > arr[hi])
lo = mid + 1;
else
hi = mid;
}

System.out.println("The number of rotations in the array is: " + lo);
}
}
