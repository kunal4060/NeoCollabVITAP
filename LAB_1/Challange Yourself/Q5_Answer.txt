import java.util.Scanner;
public class Main{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
if(sc.hasNextInt()){
int n=sc.nextInt();
int[] a=new int[n];
for(int i=0;i<n;i++)
a[i]=sc.nextInt();
int k=sc.nextInt();
for(int i=0;i<n;i+=k)
if(i+k<=n){
int left=i,right=i+k-1;
while(left<right){
int temp=a[left];
a[left]=a[right];
a[right]=temp;
left++;
right--;
}
}
for(int i=0;i<n;i++){
System.out.print(a[i]);
if(i<n-1) System.out.print(" ");
}
}
}
}
