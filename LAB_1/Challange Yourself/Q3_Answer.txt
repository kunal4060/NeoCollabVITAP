import java.util.Scanner;
public class Main{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
if(sc.hasNextInt()){
int n=sc.nextInt();
int m=sc.nextInt();
long maxRow=-1;
int maxRowIdx=-1;
long[] colSums=new long[m];
for(int i=0;i<n;i++){
long rowSum=0;
for(int j=0;j<m;j++){
long val=sc.nextLong();
rowSum+=val;
colSums[j]+=val;
}
if(rowSum>maxRow){
maxRow=rowSum;
maxRowIdx=i;
}
}
long maxCol=-1;
int maxColIdx=-1;
for(int j=0;j<m;j++)
if(colSums[j]>maxCol){
maxCol=colSums[j];
maxColIdx=j;
}
System.out.println(maxRowIdx);
System.out.println(maxColIdx);
}
}
}
