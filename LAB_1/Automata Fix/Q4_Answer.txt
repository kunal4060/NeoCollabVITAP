// You are using Java
import java.util.Scanner;

public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
int n = sc.nextInt();
int m = sc.nextInt();

for (int i = 0; i < n; i++) {
for (int j = 1; j <= m; j++) {
int val = sc.nextInt();
if (val % 2 == 0)
System.out.print(1);
else
System.out.print(0);

if (j < m)
System.out.print(" ");
}
System.out.println();
}
sc.close();
}
}
