// You are using Java
import java.util.Scanner;

class CountPrimes {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
int n = sc.nextInt();
int m = sc.nextInt();
int[][] arr = new int[n][m];
for (int i = 0; i < n; i++) {
for (int j = 0; j < m; j++) {
arr[i][j] = sc.nextInt();
}
}
int count = 0;
for (int i = 0; i < n; i++) {
for (int j = 0; j < m; j++) {
int num = arr[i][j];
if (num > 1) {
boolean isPrime = true;
for (int k = 2; k * k <= num; k++) {
if (num % k == 0) {
isPrime = false;
break;
}
}
if (isPrime) count++;
}
}
}
System.out.println(count);
}
}
