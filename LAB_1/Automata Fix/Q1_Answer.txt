// You are using Java
import java.util.Scanner;

public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);
int n = sc.nextInt();
int[][] seats = new int[n][];
int num = 1;
for (int i = 0; i < n; i++) {
seats[i] = new int[i + 1];
for (int j = 0; j <= i; j++) {
seats[i][j] = num++;
}
}
StringBuilder sb = new StringBuilder();
for (int i = 0; i < n; i++) {
for (int j = 0; j <= i; j++) {
sb.append(seats[i][j]);
if (j < i) sb.append(" ");
}
sb.append("\n");
}
System.out.print(sb);
}
}
